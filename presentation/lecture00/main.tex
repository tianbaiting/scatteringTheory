\section{散射理论的脉络}

\begin{frame}{含时演化：散射过程的物理图像}
\[
H=H_0+V,\qquad
\ket{\Psi(t)}=e^{-\ii H(t-t_0)/\hbar}\ket{\Psi(t_0)}.
\]
\begin{itemize}
\item 准备一个入射波包，观察它进入相互作用区域。
\item 波包与靶发生相互作用，随后向不同方向或通道传播。
\item 远处的探测器记录出射粒子的分布。
\end{itemize}
我们关心：给定入射态，相互作用如何改变最终测得的出射分布？
\end{frame}

\begin{frame}{含时描述：入射态与出射态}
相互作用发生前后，波包在远离散射中心的区域近似自由传播。
对于短程相互作用，用自由演化比较它的早期与晚期行为：
\[
\ket{\Psi(t)}\sim e^{-\ii H_0t/\hbar}\ket{\phi_{\mathrm{in}}}
\quad(t\to-\infty),
\]
\[
\ket{\Psi(t)}\sim e^{-\ii H_0t/\hbar}\ket{\phi_{\mathrm{out}}}
\quad(t\to+\infty).
\]
$S$ 算符联系同一次散射的入射态和出射态：
\[
\ket{\phi_{\mathrm{out}}}=S\ket{\phi_{\mathrm{in}}}.
\]
它回答：准备给定的入射态后，各种出射结果的概率幅是多少？
\end{frame}

\begin{frame}{不含时描述：按能量分解}
指数函数是时间微分算子的广义本征函数：
\[
\partial_t e^{\pm\ii\omega t}=\pm\ii\omega e^{\pm\ii\omega t}.
\]
取 $e^{-\ii\omega t}$ 约定，$\ii\hbar\partial_t$ 在该模式上作用等价于乘 $E=\hbar\omega$：
\[
\Psi_E(t,\bm r)=e^{-\ii Et/\hbar}\psi_E(\bm r),
\qquad H\psi_E=E\psi_E.
\]
\begin{itemize}
\item $H$ 不显含时间时，不同能量模式可以分别求解。
\item 一般波包是这些模式的叠加，叠加系数由初态决定。
\end{itemize}
\note{定态分离与波包叠加：\url{https://www.damtp.cam.ac.uk/user/tong/qm/qmhtml/S2.html}}
\end{frame}

\begin{frame}{定能描述：计算振幅与截面}
固定能量后，仍要指定怎样的粒子入射，以及散射波怎样离开。
以短程势的三维弹性散射为例，
\[
H\psi_E=E\psi_E,\qquad E=\frac{\hbar^2k^2}{2\mu},
\]
\[
\psi_E^{(+)}(\bm r)\sim e^{\ii kz}
+f(\hat{\bm r})\frac{e^{\ii kr}}{r},
\qquad \frac{\dd\sigma}{\dd\Omega}=|f(\hat{\bm r})|^2.
\]
\begin{itemize}
\item 含时描述说明入射与出射的意义，建立散射量的定义。
\item 定能描述把计算组织成带散射边界条件的方程。
\end{itemize}
\end{frame}

\section{求解中会遇到的技术}

\begin{frame}{受迫振动与后续散射计算}
取单位质量的受迫振子：
\[
\ddot x+\omega_0^2x=f(t),\qquad x(0)=x_0,\quad\dot x(0)=v_0.
\]
贯穿本讲的具体例子是
\[
f(t)=F_0\cos\Omega t\quad(t\ge0),\qquad
x_0=v_0=0,\qquad 0<\Omega\ne\omega_0.
\]
\begin{itemize}
\item 沿时间推进，怎样得到振子的运动轨迹？
\item 换到频率域，或把外力分解为脉冲，求解会怎样简化？
\end{itemize}
\end{frame}

\begin{frame}{同一个振子的响应}
本例的精确解同时含有驱动频率和固有频率，满足给定的零初值：
\[
x(t)=\frac{F_0}{\omega_0^2-\Omega^2}
\bigl(\cos\Omega t-\cos\omega_0t\bigr),\qquad t\ge0.
\]

\begin{frame}{沿时间推进：数值积分}
给定当前位置与速度，就能由运动方程计算加速度：
\[
\dot x=v,\qquad \dot v=f(t)-\omega_0^2x.
\]
\begin{itemize}
\item Runge--Kutta 将它作为一阶系统，在每一步估计位置和速度的变化。
\item Numerov 利用不含一阶导数的二阶方程结构，递推相邻网格点上的位移。
\item 减小步长，并与精确解比较，检查计算结果是否收敛。
\end{itemize}
\end{frame}

\begin{frame}{Fourier：哪些算子会被简化？}
把运动分解为不同频率的谐波，时间微分就变成乘法。
\[
x(t)=\int\frac{\dd\omega}{2\pi}e^{-\ii\omega t}\widetilde x(\omega),
\qquad \partial_t^2 e^{-\ii\omega t}=-\omega^2e^{-\ii\omega t}.
\]
在每个频率上，微分方程化为代数关系：
\[
(\omega_0^2-\omega^2)\widetilde x=\widetilde f,
\qquad
\widetilde x(\omega)=\frac{\widetilde f(\omega)}{\omega_0^2-\omega^2}
\quad(\omega\ne\pm\omega_0).
\]
\begin{itemize}
\item 振子的时间系数恒定，不同频率独立。
\item 本例按 $f(t)=\theta(t)F_0\cos\Omega t$ 接通外力，取 $t<0$ 静止。
\end{itemize}
\end{frame}

\begin{frame}{Laplace：变换以后初态在哪里？}
只对 $t\ge0$ 的运动作变换，可以直接追踪初值怎样进入解。
\[
X(s)=\int_0^\infty e^{-st}x(t)\dd t,\qquad
\mathcal L[\ddot x]=s^2X-sx_0-v_0.
\]
记 $F(s)=\mathcal L[f](s)$，则
\[
X(s)=\frac{F(s)+sx_0+v_0}{s^2+\omega_0^2}.
\]
初值 $x_0,v_0$ 通过分子进入解。
\end{frame}

\begin{frame}{受迫振子：升维降次后的演化方程}
把位置和速度一起作为状态，二阶方程成为一阶线性系统：
\[
\ddot x+\omega_0^2x=f(t)
\quad\Longleftrightarrow\quad
\dot{\bm y}=A\bm y+\bm b(t),
\]
\[
\bm y=\begin{pmatrix}x\\v\end{pmatrix},\qquad
A=\begin{pmatrix}0&1\\-\omega_0^2&0\end{pmatrix},\qquad
\bm b(t)=\begin{pmatrix}0\\f(t)\end{pmatrix}.
\]
\begin{itemize}
\item $A$ 描述自由运动，外力直接改变速度，再通过演化影响位置。
\item 定义 $\mathcal D=\partial_t-A$，带源方程就是 $\mathcal D\bm y=\bm b$。
\end{itemize}
线性性允许叠加不同源的响应。要从源确定唯一的运动，还需给定初始或终端数据。
\end{frame}

\begin{frame}{演化算符：传播已有的状态}
先关掉外力。自由运动满足
\[
\bm y(t)=U(t,t')\bm y(t'),\qquad
\partial_tU(t,t')=AU(t,t'),\qquad U(t',t')=I.
\]
本例中 $A$ 为常矩阵，且 $A^2=-\omega_0^2I$。令 $u=t-t'$，则
\[
U(t,t')=e^{Au}
=\begin{pmatrix}
\cos\omega_0u & \dfrac{\sin\omega_0u}{\omega_0}\\[0.5em]
-\omega_0\sin\omega_0u & \cos\omega_0u
\end{pmatrix}.
\]
\begin{itemize}
\item 两列分别给出单位初始位移、单位初始速度产生的运动。
\item $U(t,t')$ 可以向前或向后传播，尚未选择源的响应方向。
\end{itemize}
\end{frame}

\begin{frame}{带源演化：传播并叠加每次注入}
将源分解为各时刻的点源：
\[
\bm b(t)=\int_{-\infty}^{\infty}\delta(t-t')\bm b(t')\dd t'.
\]
在 $t'$ 附近注入的状态增量为 $\bm b(t')\dd t'$，
传播到 $t$ 后成为 $U(t,t')\bm b(t')\dd t'$。由线性性，
\[
\boxed{\bm y(t)=U(t,t_0)\bm y_0
+\int_{t_0}^{t}U(t,t')\bm b(t')\dd t'}.
\]
第一项传播初态，第二项累积此后各时刻的源。对第二项求导：
\[
\frac{\dd}{\dd t}\int_{t_0}^{t}U(t,t')\bm b(t')\dd t'
=\bm b(t)+A\int_{t_0}^{t}U(t,t')\bm b(t')\dd t'.
\]
上限贡献当前的源，已有的响应继续自由演化。
\end{frame}

\begin{frame}{Retarded：带时间条件的求逆核}
推迟 Green 矩阵只保留源施加之后的传播：
\[
\mathcal G_R(t,t')=\theta(t-t')U(t,t').
\]
对 $\mathcal D=\partial_t-A$，直接求导得到
\[
\begin{aligned}
\mathcal D_t\mathcal G_R(t,t')
&=\delta(t-t')U(t,t')
+\theta(t-t')(\partial_t-A)U(t,t')\\
&=\delta(t-t')I.
\end{aligned}
\]
$\mathcal G_R$ 因而是带推迟条件的求逆核。对 $t\ge t_0$，源产生的特解为
\[
\bm y_p(t)=\int_{t_0}^{t}\mathcal G_R(t,t')\bm b(t')\dd t',
\qquad \mathcal D\bm y_p=\bm b,\quad \bm y_p(t_0)=0.
\]
\end{frame}

\begin{frame}{Advanced：从终端条件向过去传播}
超前 Green 矩阵选择源时刻之前的传播：
\[
\mathcal G_A(t,t')=-\theta(t'-t)U(t,t'),\qquad
\mathcal G_A(t,t')=0\quad(t>t').
\]
由于 $\partial_t[-\theta(t'-t)]=\delta(t-t')$，它满足同一个带源方程：
\[
\mathcal D_t\mathcal G_A(t,t')=\delta(t-t')I.
\]
给定终端状态 $\bm y(T)$，对 $t\le T$ 有
\[
\bm y(t)=U(t,T)\bm y(T)
+\int_t^T\mathcal G_A(t,t')\bm b(t')\dd t'.
\]
\begin{itemize}
\item 自由项传播终端数据，积分项使用 $t$ 之后的源，并在 $T$ 为零。
\item 初值实验用推迟核，终端条件问题可用超前核向过去重建运动。
\end{itemize}
\end{frame}

\begin{frame}{标量 Green 函数与矩阵核的对应}
外力注入速度分量，我们观察位置分量，因此取矩阵核第 $1$ 行、第 $2$ 列：
\[
\begin{aligned}
G_R(t,t')&=(\mathcal G_R)_{12}
=\theta(t-t')\frac{\sin[\omega_0(t-t')]}{\omega_0},\\[0.5em]
G_A(t,t')&=(\mathcal G_A)_{12}
=-\theta(t'-t)\frac{\sin[\omega_0(t-t')]}{\omega_0}.
\end{aligned}
\]
带源演化公式的位置分量就是熟悉的受迫振动解：
\[
x(t)=x_h(t)+\int_{t_0}^{t}
\frac{\sin[\omega_0(t-t')]}{\omega_0}f(t')\dd t'.
\]
$x_h$ 是 $U(t,t_0)\bm y_0$ 的位置分量，携带初始位移和速度。
\note{自由位移为 $x_h(t)=x_0\cos[\omega_0(t-t_0)]+v_0\sin[\omega_0(t-t_0)]/\omega_0$。由于源向量的第一分量为零，位置响应只使用 Green 矩阵的第 1 行、第 2 列。}
\end{frame}

\begin{frame}{时间条件在频域中成为极点规定}
令 $u=t-t'$，取 $\widetilde G(\omega)=\int_{-\infty}^{\infty}e^{\ii\omega u}G(u)\dd u$。
对两种时间支撑分别加收敛因子 $e^{-\epsilon|u|}$，得到
\[
\begin{aligned}
\widetilde G_R(\omega)
&=\lim_{\epsilon\downarrow0}\int_0^\infty
 e^{\ii\omega u-\epsilon u}\frac{\sin\omega_0u}{\omega_0}\dd u
=\frac{1}{\omega_0^2-(\omega+\ii0^+)^2},\\[0.7em]
\widetilde G_A(\omega)
&=\lim_{\epsilon\downarrow0}\int_{-\infty}^0
 e^{\ii\omega u+\epsilon u}\frac{-\sin\omega_0u}{\omega_0}\dd u
=\frac{1}{\omega_0^2-(\omega-\ii0^+)^2}.
\end{aligned}
\]
\begin{itemize}
\item 推迟核的极点在实轴下方，超前核的极点在实轴上方。
\item 时间域选定的响应条件，在频域中由 $\pm\ii0^+$ 保留下来。
\end{itemize}
选定核后，源产生的特解由 $\widetilde x_p=\widetilde G\widetilde f$ 给出。
\end{frame}

\begin{frame}{极点绕行与因果条件}
频域中的极点规定，怎样保证外力施加之前没有响应？
\[
G_R(u)=\int_{-\infty}^{\infty}\frac{\dd\omega}{2\pi}
\frac{e^{-\ii\omega u}}{\omega_0^2-(\omega+\ii0^+)^2}.
\]
极点位于 $\omega=\pm\omega_0-\ii0^+$，都在实轴下方。
\begin{itemize}
\item $u>0$：指数在下半平面衰减，向下闭合并拾取留数。
\item $u<0$：向上闭合，内部没有极点，因此响应为零。
\end{itemize}
\[
G_R(u)=\theta(u)\frac{\sin\omega_0u}{\omega_0}.
\]
\end{frame}

\begin{frame}{Green 算符与散射边界条件}
散射方程也可以写成“线性算符作用于解等于源”：
\[
(E-H_0)\ket{\psi}=\ket{s},\qquad \ket{s}=V\ket{\psi}.
\]
给定同能量的自由入射态 $\ket{\phi}$，选择出射 Green 算符后得到
\[
G_0^{(+)}(E)=(E-H_0+\ii0^+)^{-1},\qquad
\ket{\psi^{(+)}}=\ket{\phi}+G_0^{(+)}V\ket{\psi^{(+)}}.
\]
\begin{itemize}
\item $\ket{\phi}$ 是自由入射解；$G_0^{(+)}V\ket{\psi^{(+)}}$ 是源产生的响应。
\item 散射中的 $+\ii0^+$ 规定源产生空间上的出射波：
\end{itemize}
\[
\psi^{(+)}(\bm r)\sim e^{\ii kz}+f(\hat{\bm r})\frac{e^{\ii kr}}{r}.
\]
这里的源依赖未知的 $\psi$，因此得到的是积分方程，仍需自洽求解。
\end{frame}

\begin{frame}{耦合振子与散射中的通道耦合}
两个振子相互耦合时，位移和外力成为向量。取正定的 $K$：
\[
\ddot{\bm x}+K\bm x=\bm f(t),\qquad
K=\begin{pmatrix}\omega_1^2&\lambda\\\lambda&\omega_2^2\end{pmatrix}.
\]
时间函数与振子分量构成 $\mathcal F_t\otimes\mathbb C^2$，推迟响应为
\[
\widetilde{\bm x}(\omega)
=\bigl[K-(\omega+\ii0^+)^2I\bigr]^{-1}\widetilde{\bm f}(\omega).
\]
\begin{itemize}
\item 对角化 $K$ 可以得到简正模；在原来的分量基底中仍有耦合。
\item 散射通道有类似的矩阵结构，连续的动量标签则对应积分核。
\end{itemize}
\end{frame}

\begin{frame}{时间与空间的乘积结构}
用时间函数和空间函数的乘积及其叠加组织时空依赖：
\[
\Psi(t,\bm r)=\sum_a T_a(t)\psi_a(\bm r),
\qquad \mathcal F_t\otimes\mathcal H_{\bm r}.
\]
当 $H$ 不显含时间时，方程中的算子分别作用于两个因子：
\[
\mathcal D=\ii\hbar\partial_t\otimes\id-\id\otimes H.
\]
$\mathcal F_t$ 表示时间函数空间。时间仍是演化参数，连续标签的求和用积分代替。
% [EN] Time modes organize spacetime functions, not an extra quantum subsystem / [CN] 时间模态组织时空函数，并不引入额外的量子子系统。
\note{乘积函数空间的数学背景：\url{https://www.ime.usp.br/~tausk/texts/TensorL2.pdf}}
\end{frame}

\begin{frame}{定能与分波：散射问题的变量组织}
在散射中，先分离时间模态，再用径向和角向函数组织空间：
\[
\Psi_E(t,\bm r)=e^{-\ii Et/\hbar}\psi_E(\bm r),
\]
\[
L^2(\mathbb R^3,\dd^3r)
\simeq L^2(\mathbb R_+,r^2\dd r)\otimes L^2(S^2,\dd\Omega),
\]
\[
\psi_E(\bm r)=\sum_{lm}\frac{u_{Elm}(r)}{r}Y_{lm}(\hat{\bm r}).
\]
\begin{itemize}
\item 定能模式组织时间依赖，球谐函数组织角度依赖。
\item 改用 $u=rR$ 后，径向内积的测度变成 $\dd r$。
\end{itemize}
\end{frame}

\begin{frame}{中心势中的算子分离与通道耦合}
对无自旋中心势，改用约化径向函数 $u=rR$ 后，
\[
H=h_r\otimes\id+\frac{1}{2\mu r^2}\otimes\bm L^2,
\qquad
h_r=-\frac{\hbar^2}{2\mu}\frac{\dd^2}{\dd r^2}+V(r).
\]
球谐函数使角动量算子对角化：
\[
\bm L^2Y_{lm}=\hbar^2l(l+1)Y_{lm}.
\]
\begin{itemize}
\item 每个 $l,m$ 对应一个独立的径向问题，其中保留离心项。
\item 更一般的势可耦合不同分波，得到一组耦合的径向方程。
\end{itemize}
% [EN] The radial-angular operator is written after the unitary reduction u=rR / [CN] 径向与角向算子的表达式使用了 u=rR 的幺正约化。
\note{中心势与分波：\url{https://farside.ph.utexas.edu/teaching/qm/Quantumhtml/node86.html}}
\end{frame}

\begin{frame}{径向积分与远场匹配}
取 $E=\hbar^2k^2/(2\mu)>0$，短程实中心势下的分波满足
\[
u_l''(r)+\left[k^2-\frac{l(l+1)}{r^2}
-\frac{2\mu V(r)}{\hbar^2}\right]u_l(r)=0.
\]
\begin{itemize}
\item 对原点正则的势，由 $u_l(r)\propto r^{l+1}$ 启动积分。
\item 用 Numerov 或 Runge--Kutta 沿 $r$ 推进。
\item 在远处匹配自由解，提取相移：
\end{itemize}
\[
u_l(r)\sim A_l\sin\bigl(kr-l\pi/2+\delta_l\bigr).
\]
\end{frame}

\begin{frame}{备用：直接从二阶方程构造 Green 函数}
取 $L=\partial_t^2+\omega_0^2$，直接求满足 $L_tG(t,t')=\delta(t-t')$ 的标量核。
在源点两侧，方程都是自由振动方程。令 $u=t-t'$，
\[
G(t,t')=\begin{cases}
a_-\cos\omega_0u+b_-\sin\omega_0u,&t<t',\\
a_+\cos\omega_0u+b_+\sin\omega_0u,&t>t'.
\end{cases}
\]
若 $G$ 跳跃，二阶导数会产生 $\delta'$，因此 $G$ 必须连续。
跨越源点积分 $L_tG=\delta$，得到
\[
G(t'^+,t')=G(t'^-,t'),\qquad
\partial_tG(t'^+,t')-\partial_tG(t'^-,t')=1.
\]
于是 $a_+=a_-$、$\omega_0(b_+-b_-)=1$。
再指定推迟或超前条件，才能确定所需的核。
\end{frame}

\begin{frame}{备用：二阶构造与矩阵核的对照}
推迟条件令 $t<t'$ 一侧为零。连续性和导数跳跃给出
\[
a_-=b_-=a_+=0,\qquad b_+=\frac{1}{\omega_0},
\]
\[
G_R(t,t')=\theta(t-t')\frac{\sin[\omega_0(t-t')]}{\omega_0}
=(\mathcal G_R)_{12}.
\]
超前条件令 $t>t'$ 一侧为零。同理得到
\[
a_+=b_+=a_-=0,\qquad b_-=-\frac{1}{\omega_0},
\]
\[
G_A(t,t')=-\theta(t'-t)\frac{\sin[\omega_0(t-t')]}{\omega_0}
=(\mathcal G_A)_{12}.
\]
一阶矩阵核在源点跳跃 $I$；它的 $(1,2)$ 元连续，而该元素的导数跳跃 $1$。
\end{frame}

\begin{frame}{备用：Picard 积分方程与逐次修正}
也可以把回复力放到右侧，将振子方程积分两次：
\[
x(t)=x_0+v_0t+\int_0^t(t-\tau)
\bigl[f(\tau)-\omega_0^2x(\tau)\bigr]\dd\tau.
\]
用上一轮的轨迹生成下一轮：
\[
x^{(n+1)}(t)=x_0+v_0t+\int_0^t(t-\tau)
\bigl[f(\tau)-\omega_0^2x^{(n)}(\tau)\bigr]\dd\tau.
\]
每轮用上一次的 $x$ 计算回复力，再更新整条轨迹。
\end{frame}

\begin{frame}{备用：Runge--Kutta 递推公式}
\[
\bm y=\begin{pmatrix}x\\v\end{pmatrix},\qquad
\dot{\bm y}=\bm F(t,\bm y)
=\begin{pmatrix}v\\f(t)-\omega_0^2x\end{pmatrix}.
\]
RK4 组合当前步内四次评估的斜率 $\bm k_1,\ldots,\bm k_4$：
\[
\bm y_{n+1}=\bm y_n+\frac h6
(\bm k_1+2\bm k_2+2\bm k_3+\bm k_4).
\]
\end{frame}

\begin{frame}{备用：Numerov 递推公式}
对不含一阶导数的振子方程，令
\[
t_n=nh,\qquad a=\frac{h^2\omega_0^2}{12},\qquad f_n=f(t_n).
\]
Numerov 格式直接联系三个相邻时刻：
\[
\begin{aligned}
(1+a)x_{n+1}={}&2(1-5a)x_n-(1+a)x_{n-1}\\
&+\frac{h^2}{12}\bigl(f_{n+1}+10f_n+f_{n-1}\bigr).
\end{aligned}
\]
\begin{itemize}
\item 振子由初值及 Taylor 展开启动两个点，再逐步递推。
\item 径向问题的有效系数随 $r$ 变化，需要使用相应的变系数格式。
\end{itemize}
\end{frame}

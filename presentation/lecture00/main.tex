\section{散射理论的脉络}

\begin{frame}{散射的含时图像}
\[
H=H_0+V,\qquad
\ket{\Psi(t)}=e^{-\ii H(t-t_0)/\hbar}\ket{\Psi(t_0)}.
\]
\begin{itemize}
\item 准备一个入射波包，观察它进入相互作用区域。
\item 波包与靶发生相互作用，随后向不同方向或通道传播。
\item 远处的探测器记录出射粒子的分布。
\end{itemize}
我们关心：给定入射态，相互作用如何改变最终测得的出射分布？
\end{frame}

\begin{frame}{入射态与出射态}
相互作用发生前后，波包在远离散射中心的区域近似自由传播。
对于短程相互作用，用自由演化比较它的早期与晚期行为：
\[
\ket{\Psi(t)}\sim e^{-\ii H_0t/\hbar}\ket{\phi_{\mathrm{in}}}
\quad(t\to-\infty),
\]
\[
\ket{\Psi(t)}\sim e^{-\ii H_0t/\hbar}\ket{\phi_{\mathrm{out}}}
\quad(t\to+\infty).
\]
$S$ 算符联系同一次散射的入射态和出射态：
\[
\ket{\phi_{\mathrm{out}}}=S\ket{\phi_{\mathrm{in}}}.
\]
它回答：准备给定的入射态后，各种出射结果的概率幅是多少？
\end{frame}

\begin{frame}{时空张量积与定能分解}
$H$ 不显含时间时，在时空函数的乘积空间 $\mathcal F_t\otimes\mathcal H_{\bm r}$ 中展开：
\[
\Psi=\sum_a c_a\,T_a\otimes\psi_a,\qquad
(T_a\otimes\psi_a)(t,\bm r)=T_a(t)\psi_a(\bm r).
\]
取 $T_E(t)=e^{-\ii Et/\hbar}$，则 $\ii\hbar\partial_tT_E=ET_E$。含时方程中的算子为
\[
\mathcal D=\ii\hbar\partial_t\otimes\id-\id\otimes H.
\]
对单个定能乘积解 $\Psi_E=T_E\otimes\psi_E$，
\[
\mathcal D(T_E\otimes\psi_E)=T_E\otimes(E-H)\psi_E=0.
\]
\begin{itemize}
\item 时间因子分离后，空间波函数满足 $H\psi_E=E\psi_E$。
\item 一般波包由定能乘积解叠加而成，连续能量标签用积分。
\end{itemize}
\note{$\mathcal F_t$ 表示时间函数空间，时间仍是演化参数。这里的张量积组织时空函数，谐波与散射定态按广义模态理解；标签 $a$ 可包含能量及简并指标，叠加系数由初态决定。定态分离与波包叠加：\url{https://www.damtp.cam.ac.uk/user/tong/qm/qmhtml/S2.html}}
\end{frame}

\begin{frame}{散射振幅与截面}
固定能量后，仍要指定怎样的粒子入射，以及散射波怎样离开。
以短程势的三维弹性散射为例，
\[
H\psi_E=E\psi_E,\qquad E=\frac{\hbar^2k^2}{2\mu},
\]
\[
\psi_E^{(+)}(\bm r)\sim e^{\ii kz}
+f(\hat{\bm r})\frac{e^{\ii kr}}{r},
\qquad \frac{\dd\sigma}{\dd\Omega}=|f(\hat{\bm r})|^2.
\]
\begin{itemize}
\item 含时描述说明入射与出射的意义，建立散射量的定义。
\item 定能描述把计算组织成带散射边界条件的方程。
\end{itemize}
\end{frame}

\section{求解技术}

\begin{frame}{受迫振动}
取单位质量的受迫振子：
\[
\ddot x+\omega_0^2x=f(t),\qquad x(0)=x_0,\quad\dot x(0)=v_0.
\]
贯穿本讲的具体例子是
\[
f(t)=F_0\cos\Omega t\quad(t\ge0),\qquad
x_0=v_0=0,\qquad 0<\Omega\ne\omega_0.
\]
\begin{itemize}
\item 沿时间推进，怎样得到振子的运动轨迹？
\item 换到频率域，或把外力分解为脉冲，求解会怎样简化？
\end{itemize}
\end{frame}

\begin{frame}{受迫振动的响应}
本例的精确解同时含有驱动频率和固有频率，满足给定的零初值：
\[
x(t)=\frac{F_0}{\omega_0^2-\Omega^2}
\bigl(\cos\Omega t-\cos\omega_0t\bigr),\qquad t\ge0.
\]

\begin{frame}{数值积分}
给定当前位置与速度，就能由运动方程计算加速度：
\[
\dot x=v,\qquad \dot v=f(t)-\omega_0^2x.
\]
\begin{itemize}
\item Runge--Kutta 将它作为一阶系统，在每一步估计位置和速度的变化。
\item Numerov 利用不含一阶导数的二阶方程结构，递推相邻网格点上的位移。
\item 减小步长，并与精确解比较，检查计算结果是否收敛。
\end{itemize}
\end{frame}

\begin{frame}{Fourier 变换}
把运动分解为不同频率的谐波，时间微分就变成乘法。
\[
x(t)=\int\frac{\dd\omega}{2\pi}e^{-\ii\omega t}\widetilde x(\omega),
\qquad \partial_t^2 e^{-\ii\omega t}=-\omega^2e^{-\ii\omega t}.
\]
在每个频率上，微分方程化为代数关系：
\[
(\omega_0^2-\omega^2)\widetilde x=\widetilde f,
\qquad
\widetilde x(\omega)=\frac{\widetilde f(\omega)}{\omega_0^2-\omega^2}
\quad(\omega\ne\pm\omega_0).
\]
\begin{itemize}
\item 振子的时间系数恒定，不同频率独立。
\item 本例按 $f(t)=\theta(t)F_0\cos\Omega t$ 接通外力，取 $t<0$ 静止。
\end{itemize}
\end{frame}

\begin{frame}{Laplace 变换与初值}
只对 $t\ge0$ 的运动作变换，可以直接追踪初值怎样进入解。
\[
X(s)=\int_0^\infty e^{-st}x(t)\dd t,\qquad
\mathcal L[\ddot x]=s^2X-sx_0-v_0.
\]
记 $F(s)=\mathcal L[f](s)$，则
\[
X(s)=\frac{F(s)+sx_0+v_0}{s^2+\omega_0^2}.
\]
初值 $x_0,v_0$ 通过分子进入解。
\end{frame}

\begin{frame}{振子的一阶形式}
把位置和速度一起作为状态，二阶方程成为一阶线性系统：
\[
\ddot x+\omega_0^2x=f(t)
\quad\Longleftrightarrow\quad
\dot{\bm y}=A\bm y+\bm b(t),
\]
\[
\bm y=\begin{pmatrix}x\\v\end{pmatrix},\qquad
A=\begin{pmatrix}0&1\\-\omega_0^2&0\end{pmatrix},\qquad
\bm b(t)=\begin{pmatrix}0\\f(t)\end{pmatrix}.
\]
\begin{itemize}
\item $A$ 描述自由运动，外力直接改变速度，再通过演化影响位置。
\item 定义 $\mathcal D=\partial_t-A$，带源方程就是 $\mathcal D\bm y=\bm b$。
\end{itemize}
线性性允许叠加不同源的响应。要从源确定唯一的运动，还需给定初始或终端数据。
\end{frame}

\begin{frame}{演化算符与 Green 核}
演化算符 $U$ 将已有状态按自由方程传播：
\[
\bm y(t)=U(t,t')\bm y(t'),\qquad
(\partial_t-A)U(t,t')=0,\qquad U(t',t')=I.
\]
本例中 $A^2=-\omega_0^2I$，因此
\[
U(t,t')=e^{A(t-t')}
=\begin{pmatrix}
\cos[\omega_0(t-t')] & \dfrac{\sin[\omega_0(t-t')]}{\omega_0}\\[0.5em]
-\omega_0\sin[\omega_0(t-t')] & \cos[\omega_0(t-t')]
\end{pmatrix}.
\]
两列分别给出单位初始位移、单位初始速度产生的自由运动。
\medskip

只保留源施加之后的传播，就得到推迟 Green 核
$\mathcal G_R(t,t')=\theta(t-t')U(t,t')$。
\end{frame}

\begin{frame}{单位冲量响应}
振子原先静止。在 $t'$ 时刻施加单位冲量 $f(t)=\delta(t-t')$，
位置连续，速度增加 $1$：
\[
x(t'^+)-x(t'^-)=0,\qquad v(t'^+)-v(t'^-)=1.
\]
此后没有外力，注入的状态增量按 $U(t,t')$ 自由演化：
\[
\bm y_{\mathrm{pulse}}(t)
=\theta(t-t')U(t,t')\begin{pmatrix}0\\1\end{pmatrix}
=\theta(t-t')\begin{pmatrix}
\dfrac{\sin[\omega_0(t-t')]}{\omega_0}\\[0.5em]
\cos[\omega_0(t-t')]
\end{pmatrix}.
\]
\end{frame}

\begin{frame}{推迟条件}
固定源施加时刻 $t'$，$t$ 表示观察响应的时刻。
在 $t'$ 注入一个状态增量 $\bm q$，对应的源与响应为
\[
\bm b(t)=\delta(t-t')\bm q,\qquad
\bm y_{\mathrm{pulse}}(t)=\mathcal G_R(t,t')\bm q.
\]
\textbf{推迟（retarded）条件：这次脉冲引起的响应不早于脉冲本身。}
因此，对任意 $\bm q$，
\[
t<t'\quad\Longrightarrow\quad
\bm y_{\mathrm{pulse}}(t)=0
\quad\Longrightarrow\quad \mathcal G_R(t,t')=0.
\]
这里为零的是这次脉冲的响应；初态产生的自由运动另计。
阶跃函数将源施加前后区分开：
\[
\theta(t-t')=\begin{cases}0,&t<t',\\1,&t>t'.\end{cases}
\]
\end{frame}

\begin{frame}{Green 算符与求逆}
要求逆的是作用于整个状态向量的矩阵微分算符：
\[
\mathcal D=\partial_t I-A
=\begin{pmatrix}\partial_t&-1\\\omega_0^2&\partial_t\end{pmatrix},
\qquad \mathcal D\bm y=\bm b.
\]
其推迟 Green 矩阵满足
\[
\mathcal D_t\mathcal G_R(t,t')=\delta(t-t')I,
\qquad \mathcal G_R(t,t')=0\quad(t<t').
\]
$\delta(t-t')I$ 是恒等算符的积分核，对应于矩阵求逆中的 $MM^{-1}=I$。
\medskip

将 $\mathcal D$ 限制在满足 $\bm y_p(t_0)=0$ 的函数上，记为 $\mathcal D_R$。
以 $\mathcal G_R(t,t')$ 为核的积分算符就是它的逆：
\[
\boxed{\widehat{\mathcal G}_R=\mathcal D_R^{-1}},\qquad
\bm y_p=\widehat{\mathcal G}_R\bm b.
\]
\end{frame}

\begin{frame}{源时刻的跳跃条件}
固定 $t'$，对 $\mathcal D_t\mathcal G_R=\delta(t-t')I$
在观察时刻 $t\in[t'-\epsilon,t'+\epsilon]$ 上积分：
\[
\int_{t'-\epsilon}^{t'+\epsilon}
\bigl[\partial_t\mathcal G_R(t,t')-A\mathcal G_R(t,t')\bigr]\dd t=I.
\]
$t'^-$ 表示从源施加之前趋近 $t'$，$t'^+$ 表示从源施加之后趋近：
\[
\mathcal G_R(t'^\pm,t')
=\lim_{\epsilon\downarrow0}\mathcal G_R(t'\pm\epsilon,t').
\]
当 $\epsilon\to0$ 时，有限的 $A\mathcal G_R$ 在缩小区间上的积分趋于零。
导数项留下前后两个极限的差：
\[
\boxed{\mathcal G_R(t'^+,t')-\mathcal G_R(t'^-,t')=I}.
\]
单位矩阵源使 Green 矩阵在源施加时刻跳跃 $I$。
\end{frame}

\begin{frame}{推迟 Green 矩阵}
推迟条件给出源施加之前的极限，跳跃条件确定施加之后的初值：
\[
\mathcal G_R(t'^-,t')=0,\qquad
\mathcal G_R(t'^+,t')=I.
\]
单位脉冲只在 $t'$ 作用。$t>t'$ 时，矩阵按齐次方程演化：
\[
\partial_t\mathcal G_R=A\mathcal G_R,\qquad
\mathcal G_R(t,t')=U(t,t')\mathcal G_R(t'^+,t')=e^{A(t-t')}.
\]
结合 $t<t'$ 时为零的条件，得到
\[
\mathcal G_R(t,t')=\theta(t-t')\begin{pmatrix}
\cos[\omega_0(t-t')] & \dfrac{\sin[\omega_0(t-t')]}{\omega_0}\\[0.5em]
-\omega_0\sin[\omega_0(t-t')] & \cos[\omega_0(t-t')]
\end{pmatrix}.
\]
\end{frame}

\begin{frame}{源项叠加}
将源分解为各时刻的点源：
\[
\bm b(t)=\int_{-\infty}^{\infty}\delta(t-t')\bm b(t')\dd t'.
\]
每次注入 $\bm b(t')\dd t'$，产生响应
$\mathcal G_R(t,t')\bm b(t')\dd t'$。由线性性，
\[
\bm y_p(t)=(\widehat{\mathcal G}_R\bm b)(t)
=\int_{t_0}^{t}\mathcal G_R(t,t')\bm b(t')\dd t'.
\]
对它作用 $\mathcal D$，就还原出源：
\[
\mathcal D\bm y_p=\bm b,\qquad \bm y_p(t_0)=0.
\]
非零初态由自由演化携带，完整解为
\[
\boxed{\begin{pmatrix}x(t)\\v(t)\end{pmatrix}
=U(t,t_0)\begin{pmatrix}x_0\\v_0\end{pmatrix}
+\int_{t_0}^{t}\mathcal G_R(t,t')
\begin{pmatrix}0\\f(t')\end{pmatrix}\dd t'}.
\]
\end{frame}

\begin{frame}{受迫振动解}
取 $t_0=0$、$\bm y_0=0$，在 $t\ge0$ 施加 $f(t)=F_0\cos\Omega t$，
其中 $0<\Omega\ne\omega_0$。将整个源向量代入：
\[
\begin{aligned}
\begin{pmatrix}x(t)\\v(t)\end{pmatrix}
&=\int_0^t\mathcal G_R(t,t')
\begin{pmatrix}0\\F_0\cos\Omega t'\end{pmatrix}\dd t'\\[0.4em]
&=F_0\int_0^t\begin{pmatrix}
\dfrac{\sin[\omega_0(t-t')]}{\omega_0}\cos\Omega t'\\[0.5em]
\cos[\omega_0(t-t')]\cos\Omega t'
\end{pmatrix}\dd t'.
\end{aligned}
\]
计算积分，得到完整的状态：
\[
\boxed{\begin{pmatrix}x(t)\\v(t)\end{pmatrix}
=\frac{F_0}{\omega_0^2-\Omega^2}
\begin{pmatrix}
\cos\Omega t-\cos\omega_0t\\
-\Omega\sin\Omega t+\omega_0\sin\omega_0t
\end{pmatrix}}.
\]
第一分量就是前面展示的位移曲线，第二分量给出速度。
\end{frame}

\begin{frame}{超前 Green 矩阵}
若从终端状态向过去重建运动，选择超前（advanced）核：
\[
\mathcal G_A(t,t')=-\theta(t'-t)U(t,t'),\qquad
\mathcal D_t\mathcal G_A(t,t')=\delta(t-t')I.
\]
它在 $t>t'$ 为零，并在源点满足同样的矩阵跳跃条件：
\[
\mathcal G_A(t'^+,t')-\mathcal G_A(t'^-,t')=0-(-I)=I.
\]
给定终端状态 $\bm y(T)$，对 $t\le T$ 有
\[
\bm y(t)=U(t,T)\bm y(T)
+\int_t^T\mathcal G_A(t,t')\bm b(t')\dd t'.
\]
\begin{itemize}
\item 积分项在 $T$ 为零，给出 $\mathcal D$ 在零终端数据下的逆作用。
\item 推迟与超前核求逆的是同一个微分表达式，指定的时间条件不同。
\end{itemize}
\note{因为 $\partial_t[-\theta(t'-t)]=\delta(t-t')$，超前核的负号保证单位源的符号与推迟核一致。}
\end{frame}

\begin{frame}{频域矩阵求逆}
令 $\tau=t-t'$，取 Fourier 约定 $e^{-\ii\omega t}$。
时间微分变为 $-\ii\omega$，于是
\[
\widetilde{\mathcal D}(\omega)=-\ii\omega I-A
=\begin{pmatrix}-\ii\omega&-1\\\omega_0^2&-\ii\omega\end{pmatrix}.
\]
推迟核的变换用 $e^{-\epsilon\tau}$ 保证收敛，得到
\[
\begin{aligned}
\widetilde{\mathcal G}_R(\omega)
&=\bigl[-\ii(\omega+\ii0^+)I-A\bigr]^{-1}\\[0.5em]
&=\frac{1}{\omega_0^2-(\omega+\ii0^+)^2}
\begin{pmatrix}
-\ii(\omega+\ii0^+)&1\\
-\omega_0^2&-\ii(\omega+\ii0^+)
\end{pmatrix}.
\end{aligned}
\]
超前核则为 $\widetilde{\mathcal G}_A=[-\ii(\omega-\ii0^+)I-A]^{-1}$。
\par
推迟极点在下半平面，超前极点在上半平面。
\note{对推迟核，$\widetilde{\mathcal G}_R(\omega)=\lim_{\epsilon\downarrow0}\int_0^\infty e^{(\ii\omega-\epsilon)\tau}e^{A\tau}\dd\tau=[-\ii(\omega+\ii0^+)I-A]^{-1}$。超前核在负时间差上积分，收敛因子为 $e^{\epsilon\tau}$，对应另一侧的极点规定。}
\end{frame}

\begin{frame}{极点与时间支撑}
对完整的矩阵逐元作 Fourier 逆变换：
\[
\mathcal G_R(t,t')=\int_{-\infty}^{\infty}
\frac{\dd\omega}{2\pi}\,
 e^{-\ii\omega(t-t')}\widetilde{\mathcal G}_R(\omega).
\]
矩阵元的极点均在 $\omega=\pm\omega_0-\ii0^+$，位于实轴下方。
\begin{itemize}
\item $t>t'$：向下闭合，留数给出 $e^{A(t-t')}$。
\item $t<t'$：向上闭合，内部没有极点，整个矩阵核为零。
\end{itemize}
\[
\mathcal G_R(t,t')=\theta(t-t')e^{A(t-t')}.
\]
超前核的极点位于上半平面，对应
\[
\mathcal G_A(t,t')=-\theta(t'-t)e^{A(t-t')}.
\]
\end{frame}

\begin{frame}{散射 Green 算符}
散射方程也可以写成“线性算符作用于解等于源”：
\[
(E-H_0)\ket{\psi}=\ket{s},\qquad \ket{s}=V\ket{\psi}.
\]
给定同能量的自由入射态 $\ket{\phi}$，选择出射 Green 算符后得到
\[
G_0^{(+)}(E)=(E-H_0+\ii0^+)^{-1},\qquad
\ket{\psi^{(+)}}=\ket{\phi}+G_0^{(+)}V\ket{\psi^{(+)}}.
\]
\begin{itemize}
\item $\ket{\phi}$ 是自由入射解；$G_0^{(+)}V\ket{\psi^{(+)}}$ 是源产生的响应。
\item 散射中的 $+\ii0^+$ 规定源产生空间上的出射波：
\end{itemize}
\[
\psi^{(+)}(\bm r)\sim e^{\ii kz}+f(\hat{\bm r})\frac{e^{\ii kr}}{r}.
\]
这里的源依赖未知的 $\psi$，因此得到的是积分方程，仍需自洽求解。
\end{frame}

\begin{frame}{耦合振子与散射通道}
两个振子相互耦合时，位移和外力成为向量。取正定的 $K$：
\[
\ddot{\bm x}+K\bm x=\bm f(t),\qquad
K=\begin{pmatrix}\omega_1^2&\lambda\\\lambda&\omega_2^2\end{pmatrix}.
\]
时间函数与振子分量构成 $\mathcal F_t\otimes\mathbb C^2$，推迟响应为
\[
\widetilde{\bm x}(\omega)
=\bigl[K-(\omega+\ii0^+)^2I\bigr]^{-1}\widetilde{\bm f}(\omega).
\]
\begin{itemize}
\item 对角化 $K$ 可以得到简正模；在原来的分量基底中仍有耦合。
\item 散射通道有类似的矩阵结构，连续的动量标签则对应积分核。
\end{itemize}
\end{frame}

\begin{frame}{时空乘积结构}
用时间函数和空间函数的乘积及其叠加组织时空依赖：
\[
\Psi(t,\bm r)=\sum_a T_a(t)\psi_a(\bm r),
\qquad \mathcal F_t\otimes\mathcal H_{\bm r}.
\]
当 $H$ 不显含时间时，方程中的算子分别作用于两个因子：
\[
\mathcal D=\ii\hbar\partial_t\otimes\id-\id\otimes H.
\]
$\mathcal F_t$ 表示时间函数空间。时间仍是演化参数，连续标签的求和用积分代替。
% [EN] Time modes organize spacetime functions, not an extra quantum subsystem / [CN] 时间模态组织时空函数，并不引入额外的量子子系统。
\note{乘积函数空间的数学背景：\url{https://www.ime.usp.br/~tausk/texts/TensorL2.pdf}}
\end{frame}

\begin{frame}{定能与分波展开}
在散射中，先分离时间模态，再用径向和角向函数组织空间：
\[
\Psi_E(t,\bm r)=e^{-\ii Et/\hbar}\psi_E(\bm r),
\]
\[
L^2(\mathbb R^3,\dd^3r)
\simeq L^2(\mathbb R_+,r^2\dd r)\otimes L^2(S^2,\dd\Omega),
\]
\[
\psi_E(\bm r)=\sum_{lm}\frac{u_{Elm}(r)}{r}Y_{lm}(\hat{\bm r}).
\]
\begin{itemize}
\item 定能模式组织时间依赖，球谐函数组织角度依赖。
\item 改用 $u=rR$ 后，径向内积的测度变成 $\dd r$。
\end{itemize}
\end{frame}

\begin{frame}{中心势与分波解耦}
对无自旋中心势，改用约化径向函数 $u=rR$ 后，
\[
H=h_r\otimes\id+\frac{1}{2\mu r^2}\otimes\bm L^2,
\qquad
h_r=-\frac{\hbar^2}{2\mu}\frac{\dd^2}{\dd r^2}+V(r).
\]
球谐函数使角动量算子对角化：
\[
\bm L^2Y_{lm}=\hbar^2l(l+1)Y_{lm}.
\]
\begin{itemize}
\item 每个 $l,m$ 对应一个独立的径向问题，其中保留离心项。
\item 更一般的势可耦合不同分波，得到一组耦合的径向方程。
\end{itemize}
% [EN] The radial-angular operator is written after the unitary reduction u=rR / [CN] 径向与角向算子的表达式使用了 u=rR 的幺正约化。
\note{中心势与分波：\url{https://farside.ph.utexas.edu/teaching/qm/Quantumhtml/node86.html}}
\end{frame}

\begin{frame}{径向积分与远场匹配}
取 $E=\hbar^2k^2/(2\mu)>0$，短程实中心势下的分波满足
\[
u_l''(r)+\left[k^2-\frac{l(l+1)}{r^2}
-\frac{2\mu V(r)}{\hbar^2}\right]u_l(r)=0.
\]
\begin{itemize}
\item 对原点正则的势，由 $u_l(r)\propto r^{l+1}$ 启动积分。
\item 用 Numerov 或 Runge--Kutta 沿 $r$ 推进。
\item 在远处匹配自由解，提取相移：
\end{itemize}
\[
u_l(r)\sim A_l\sin\bigl(kr-l\pi/2+\delta_l\bigr).
\]
\end{frame}

\begin{frame}{附录：二阶算符求逆}
从原始受迫振动方程出发：
\[
Lx=f,\qquad L=\partial_t^2+\omega_0^2.
\]
标量 Green 函数由单位点源方程定义：
\[
(\partial_t^2+\omega_0^2)G(t,t')=\delta(t-t').
\]
对于源施加前没有响应的推迟解，另加条件
\[
G_R(t,t')=0\qquad(t<t').
\]
将 $L$ 限制在零初值 $x_p(t_0)=\dot x_p(t_0)=0$ 上，记为 $L_R$。
\par
Green 核给出其逆的积分表示：
\[
x_p(t)=(L_R^{-1}f)(t)
=\int_{t_0}^{t}G_R(t,t')f(t')\dd t'.
\]
\note{一般初值时，再加自由解 $x_h(t)=x_0\cos[\omega_0(t-t_0)]+v_0\sin[\omega_0(t-t_0)]/\omega_0$。本附录直接求解二阶算符的 Green 函数。}
\end{frame}

\begin{frame}{附录：标量核的跳跃条件}
在 $t\ne t'$ 时，$L_tG=0$，所以源时刻前后的解分别为
\[
G(t,t')=\begin{cases}
a_-\cos[\omega_0(t-t')]+b_-\sin[\omega_0(t-t')],&t<t',\\
a_+\cos[\omega_0(t-t')]+b_+\sin[\omega_0(t-t')],&t>t'.
\end{cases}
\]
若 $G$ 跳跃，二阶导数会产生 $\delta'$。源中只有 $\delta$，因此 $G$ 必须连续。
跨越 $t'$ 积分原方程：
\[
\int_{t'-\epsilon}^{t'+\epsilon}
\bigl[\partial_t^2G(t,t')+\omega_0^2G(t,t')\bigr]\dd t=1.
\]
取 $\epsilon\downarrow0$，得到
\[
G(t'^+,t')=G(t'^-,t'),\qquad
\partial_tG(t'^+,t')-\partial_tG(t'^-,t')=1.
\]
所以 $a_+=a_-$，且 $\omega_0(b_+-b_-)=1$。
\end{frame}

\begin{frame}{附录：推迟与超前标量核}
推迟条件要求 $t<t'$ 时 $G_R(t,t')=0$，即 $a_-=b_-=0$。
连续性和导数跳跃进一步给出
\[
a_+=0,\qquad b_+=\frac{1}{\omega_0},
\]
\[
\boxed{G_R(t,t')=\theta(t-t')\frac{\sin[\omega_0(t-t')]}{\omega_0}}.
\]
超前条件要求 $t>t'$ 时 $G_A(t,t')=0$，即 $a_+=b_+=0$。
同样由连续性和导数跳跃得到
\[
a_-=0,\qquad b_-=-\frac{1}{\omega_0},
\]
\[
\boxed{G_A(t,t')=-\theta(t'-t)\frac{\sin[\omega_0(t-t')]}{\omega_0}}.
\]
\end{frame}

\begin{frame}{附录：Picard 迭代}
也可以把回复力放到右侧，将振子方程积分两次：
\[
x(t)=x_0+v_0t+\int_0^t(t-\tau)
\bigl[f(\tau)-\omega_0^2x(\tau)\bigr]\dd\tau.
\]
用上一轮的轨迹生成下一轮：
\[
x^{(n+1)}(t)=x_0+v_0t+\int_0^t(t-\tau)
\bigl[f(\tau)-\omega_0^2x^{(n)}(\tau)\bigr]\dd\tau.
\]
每轮用上一次的 $x$ 计算回复力，再更新整条轨迹。
\end{frame}

\begin{frame}{附录：Runge--Kutta 格式}
\[
\bm y=\begin{pmatrix}x\\v\end{pmatrix},\qquad
\dot{\bm y}=\bm F(t,\bm y)
=\begin{pmatrix}v\\f(t)-\omega_0^2x\end{pmatrix}.
\]
RK4 组合当前步内四次评估的斜率 $\bm k_1,\ldots,\bm k_4$：
\[
\bm y_{n+1}=\bm y_n+\frac h6
(\bm k_1+2\bm k_2+2\bm k_3+\bm k_4).
\]
\end{frame}

\begin{frame}{附录：Numerov 格式}
对不含一阶导数的振子方程，令
\[
t_n=nh,\qquad a=\frac{h^2\omega_0^2}{12},\qquad f_n=f(t_n).
\]
Numerov 格式直接联系三个相邻时刻：
\[
\begin{aligned}
(1+a)x_{n+1}={}&2(1-5a)x_n-(1+a)x_{n-1}\\
&+\frac{h^2}{12}\bigl(f_{n+1}+10f_n+f_{n-1}\bigr).
\end{aligned}
\]
\begin{itemize}
\item 振子由初值及 Taylor 展开启动两个点，再逐步递推。
\item 径向问题的有效系数随 $r$ 变化，需要使用相应的变系数格式。
\end{itemize}
\end{frame}
